# Bu script RTA4004 cihazındaki bus table dan çekilen dataları analiz eder. import os import glob import time import sys # Kütüphane kontrolü try: import pandas as pd import matplotlib.pyplot as plt except ImportError: print("HATA: 'pandas' veya 'matplotlib' kütüphanesi yüklü değil.") input("\nÇıkmak için Enter'a basın...") sys.exit() def parse_t100_verified(hex_str): """ 16-bit Little Endian Parsing (Örn: 3690D1075C08...) X: Byte 2 (LSB), Byte 3 (MSB) -> D1 07 -> 0x07D1 (2001) Y: Byte 4 (LSB), Byte 5 (MSB) -> 5C 08 -> 0x085C (2140) """ try: data = bytes.fromhex(hex_str.replace("0x", "").replace(" ", "")) if len(data) < 7: return None status = data[1] x = data[2] | (data[3] << 8) y = data[4] | (data[5] << 8) pressure = data[6] return { 'id': data[0], 'x': x, 'y': y, 'p': pressure, 'detect': bool(status & 0x80), # Bit 7: Detect 'rel': bool(status & 0x20) # Bit 5: Release } except: return None def main(): script_dir = os.path.dirname(os.path.abspath(__file__)) os.chdir(script_dir) csv_files = [] for ext in ["*.csv", "*.CSV"]: csv_files.extend(glob.glob(ext)) csv_files = list(set(csv_files)) if not csv_files: print(f"\nHATA: CSV bulunamadı!") else: output_txt = "touch_analiz_sonuclari.txt" with open(output_txt, "w", encoding="utf-8") as f_out: f_out.write(f"ATMEL MAXTOUCH T100 ANALİZ RAPORU\n") for file in csv_files: print(f"İşleniyor: {file}...") try: df = pd.read_csv(file, skiprows=2) # SÜTUN BAŞLIĞI DÜZELTMESİ: # 'Start Time in s' veya 'Time Diff. in s' hangisi varsa onu kullan time_col = None for col in ['Start Time in s', 'Time Diff. in s']: if col in df.columns: time_col = col break if not time_col: print(f" -> HATA: Zaman sütunu bulunamadı! Mevcut sütunlar: {df.columns.tolist()}") continue f_out.write(f"\nDOSYA: {file}\n" + "-"*40 + "\n") points_x, points_y = [], [] for _, row in df.iterrows(): if str(row['Type']).strip() == 'Read' and '0x' in str(row['Data']): p = parse_t100_verified(str(row['Data'])) if p: t = float(row[time_col]) durum = "DOKUNMA" if p['detect'] else "IDLE" f_out.write(f"{t:<12.6f} | X:{p['x']:<4} | Y:{p['y']:<4} | {durum}\n") if p['detect']: points_x.append(p['x']) points_y.append(p['y']) # Görselleştirme if points_x: plt.figure(figsize=(10, 8)) plt.xlim(0, 4095) plt.ylim(0, 4095) plt.gca().invert_yaxis() # Origin Sol Üst plt.plot(points_x, points_y, 'b-o', markersize=2, alpha=0.5) plt.title(f"Simülasyon: {file}") plt.savefig(f"{os.path.splitext(file)[0]}_simulasyon.png") plt.close() except Exception as e: print(f" -> Hata: {file} işlenemedi: {e}") print(f"\nİşlem tamamlandı. ENTER'a basın.") input() if __name__ == "__main__": main()
